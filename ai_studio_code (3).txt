// 1. PASSWORD VAULT & GENERATOR
// ==========================================
class PasswordVaultScreen extends StatefulWidget {
  const PasswordVaultScreen({super.key});

  @override
  State<PasswordVaultScreen> createState() => _PasswordVaultScreenState();
}

class _PasswordVaultScreenState extends State<PasswordVaultScreen> {
  final List<Map<String, String>> _vault = [];
  final _serviceController = TextEditingController();
  final _secretController = TextEditingController();
  double _passwordLength = 16;
  bool _includeSymbols = true;
  String _generatedPassword = '';

  void _generatePassword() {
    const letters = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ';
    const numbers = '0123456789';
    const symbols = '!@#\$%^&*()-_=+[]{}|;:<>?';
    String chars = letters + numbers + (_includeSymbols ? symbols : '');

    final random = Random.secure();
    setState(() {
      _generatedPassword = List.generate(
        _passwordLength.round(),
        (index) => chars[random.nextInt(chars.length)],
      ).join();
      _secretController.text = _generatedPassword;
    });
  }

  void _saveEntry() {
    if (_serviceController.text.isNotEmpty && _secretController.text.isNotEmpty) {
      setState(() {
        _vault.add({
          'service': _serviceController.text,
          'secret': _secretController.text,
        });
        _serviceController.clear();
        _secretController.clear();
      });
    }
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text('Credential Vault & Generator', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
          const SizedBox(height: 12),
          Card(
            child: Padding(
              padding: const EdgeInsets.all(12.0),
              child: Column(
                children: [
                  TextField(
                    controller: _serviceController,
                    decoration: const InputDecoration(labelText: 'Service Name (e.g. AWS, Internal API)'),
                  ),
                  TextField(
                    controller: _secretController,
                    obscureText: true,
                    decoration: const InputDecoration(labelText: 'Secret / Password'),
                  ),
                  Row(
                    children: [
                      Text('Length: ${_passwordLength.round()}'),
                      Expanded(
                        child: Slider(
                          value: _passwordLength,
                          min: 8,
                          max: 32,
                          divisions: 24,
                          onChanged: (val) => setState(() => _passwordLength = val),
                        ),
                      ),
                      IconButton(
                        icon: const Icon(Icons.refresh, color: Color(0xFF00FF66)),
                        onPressed: _generatePassword,
                      ),
                    ],
                  ),
                  ElevatedButton.icon(
                    onPressed: _saveEntry,
                    icon: const Icon(Icons.save),
                    label: const Text('Store in Vault'),
                  ),
                ],
              ),
            ),
          ),
          const SizedBox(height: 16),
          const Text('Secure Records', style: TextStyle(fontWeight: FontWeight.bold)),
          Expanded(
            child: _vault.isEmpty
                ? const Center(child: Text('No credentials currently stored.'))
                : ListView.builder(
                    itemCount: _vault.length,
                    itemBuilder: (context, idx) {
                      return Card(
                        child: ListTile(
                          leading: const Icon(Icons.key, color: Color(0xFF58A6FF)),
                          title: Text(_vault[idx]['service']!),
                          subtitle: Text('••••••••••••• (${_vault[idx]['secret']!.length} chars)'),
                          trailing: IconButton(
                            icon: const Icon(Icons.delete, color: Colors.red),
                            onPressed: () => setState(() => _vault.removeAt(idx)),
                          ),
                        ),
                      );
                    },
                  ),
          ),
        ],
      ),
    );
  }
}

// 2. NETWORK AUDITOR & PORT SCANNER
// ==========================================
class NetworkAuditorScreen extends StatefulWidget {
  const NetworkAuditorScreen({super.key});

  @override
  State<NetworkAuditorScreen> createState() => _NetworkAuditorScreenState();
}

class _NetworkAuditorScreenState extends State<NetworkAuditorScreen> {
  final _hostController = TextEditingController(text: '127.0.0.1');
  bool _isScanning = false;
  final List<Map<String, dynamic>> _results = [];

  final Map<int, String> _targetPorts = {
    21: 'FTP (Insecure)',
    22: 'SSH',
    23: 'Telnet (Cleartext)',
    80: 'HTTP',
    443: 'HTTPS',
    445: 'SMB',
    3306: 'MySQL',
    3389: 'RDP',
    8080: 'HTTP-Alt',
  };

  Future<void> _runAudit() async {
    setState(() {
      _isScanning = true;
      _results.clear();
    });

    final host = _hostController.text.trim();
    for (final entry in _targetPorts.entries) {
      if (!_isScanning) break;
      final port = entry.key;
      final desc = entry.value;

      bool isOpen = false;
      try {
        final socket = await Socket.connect(host, port, timeout: const Duration(milliseconds: 400));
        socket.destroy();
        isOpen = true;
      } catch (_) {
        isOpen = false;
      }

      setState(() {
        _results.add({'port': port, 'desc': desc, 'open': isOpen});
      });
    }

    setState(() => _isScanning = false);
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text('Authorized Port & Host Auditor', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
          const SizedBox(height: 12),
          Row(
            children: [
              Expanded(
                child: TextField(
                  controller: _hostController,
                  decoration: const InputDecoration(labelText: 'Target Host IP', border: OutlineInputBorder()),
                ),
              ),
              const SizedBox(width: 8),
              ElevatedButton(
                onPressed: _isScanning ? () => setState(() => _isScanning = false) : _runAudit,
                style: ElevatedButton.styleFrom(
                  backgroundColor: _isScanning ? Colors.red : const Color(0xFF00FF66),
                  foregroundColor: Colors.black,
                ),
                child: Text(_isScanning ? 'Stop' : 'Scan'),
              ),
            ],
          ),
          const SizedBox(height: 16),
          if (_isScanning) const LinearProgressIndicator(),
          Expanded(
            child: ListView.builder(
              itemCount: _results.length,
              itemBuilder: (context, idx) {
                final item = _results[idx];
                final isOpen = item['open'] as bool;
                return Card(
                  child: ListTile(
                    leading: Icon(
                      isOpen ? Icons.lock_open : Icons.lock_outline,
                      color: isOpen ? Colors.redAccent : Colors.green,
                    ),
                    title: Text('Port ${item['port']} - ${item['desc']}'),
                    trailing: Text(
                      isOpen ? 'OPEN' : 'CLOSED',
                      style: TextStyle(
                        color: isOpen ? Colors.redAccent : Colors.green,
                        fontWeight: FontWeight.bold,
                      ),
                    ),
                  ),
                );
              },
            ),
          ),
        ],
      ),
    );
  }
}

// 3. PHISHING HEURISTIC DETECTOR
// ==========================================
class PhishingDetectorScreen extends StatefulWidget {
  const PhishingDetectorScreen({super.key});

  @override
  State<PhishingDetectorScreen> createState() => _PhishingDetectorScreenState();
}

class _PhishingDetectorScreenState extends State<PhishingDetectorScreen> {
  final _urlController = TextEditingController();
  List<String> _flags = [];
  double _riskScore = 0.0;
  bool _evaluated = false;

  void _analyzeUrl() {
    final rawUrl = _urlController.text.trim();
    final List<String> flags = [];
    double score = 0.0;

    Uri? uri;
    try {
      uri = Uri.parse(rawUrl.contains('://') ? rawUrl : 'https://$rawUrl');
    } catch (_) {
      setState(() {
        _flags = ['Malformed URL syntax'];
        _riskScore = 1.0;
        _evaluated = true;
      });
      return;
    }

    final host = uri.host.toLowerCase();

    // Heuristic 1: Raw IP check
    if (RegExp(r'^(\d{1,3}\.){3}\d{1,3}$').hasMatch(host)) {
      flags.add('Host uses a raw numerical IP address instead of domain.');
      score += 0.45;
    }
    // Heuristic 2: Punycode / IDN spoofing
    if (host.startsWith('xn--')) {
      flags.add('Punycode detected: Target may spoof legitimate glyphs.');
      score += 0.5;
    }
    // Heuristic 3: Subdomain chaining
    if (host.split('.').length > 4) {
      flags.add('Excessive subdomains (common spoofing disguise).');
      score += 0.25;
    }
    // Heuristic 4: Embedded auth credentials
    if (uri.userInfo.isNotEmpty) {
      flags.add('Embedded user info (@ symbol used to hide real destination).');
      score += 0.6;
    }
    // Heuristic 5: Insecure cleartext HTTP
    if (uri.scheme == 'http') {
      flags.add('Uses insecure cleartext HTTP protocol.');
      score += 0.2;
    }

    setState(() {
      _flags = flags;
      _riskScore = score.clamp(0.0, 1.0);
      _evaluated = true;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text('Phishing URL Heuristic Inspector', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
          const SizedBox(height: 12),
          TextField(
            controller: _urlController,
            decoration: const InputDecoration(
              labelText: 'URL to inspect (e.g. http://paypal.com@192.168.1.1)',
              border: OutlineInputBorder(),
            ),
          ),
          const SizedBox(height: 10),
          ElevatedButton(
            onPressed: _analyzeUrl,
            child: const Text('Analyze Link Risk'),
          ),
          const SizedBox(height: 20),
          if (_evaluated) ...[
            Text('Calculated Threat Score: ${(_riskScore * 100).toInt()}%',
                style: TextStyle(
                  fontSize: 16,
                  fontWeight: FontWeight.bold,
                  color: _riskScore > 0.4 ? Colors.redAccent : Colors.green,
                )),
            LinearProgressIndicator(
              value: _riskScore,
              color: _riskScore > 0.4 ? Colors.redAccent : Colors.green,
              backgroundColor: Colors.grey[800],
            ),
            const SizedBox(height: 16),
            const Text('Heuristic Indicators:', style: TextStyle(fontWeight: FontWeight.bold)),
            Expanded(
              child: _flags.isEmpty
                  ? const Center(child: Text('No common phishing anomalies found.'))
                  : ListView.builder(
                      itemCount: _flags.length,
                      itemBuilder: (context, idx) => ListTile(
                        leading: const Icon(Icons.warning_amber_rounded, color: Colors.orangeAccent),
                        title: Text(_flags[idx]),
                      ),
                    ),
            ),
          ]
        ],
      ),
    );
  }
}

// 4. CRYPTO TOOL (AES-256 & HASHING)
// ==========================================
class CryptoToolScreen extends StatefulWidget {
  const CryptoToolScreen({super.key});

  @override
  State<CryptoToolScreen> createState() => _CryptoToolScreenState();
}

class _CryptoToolScreenState extends State<CryptoToolScreen> {
  final _inputController = TextEditingController();
  final _keyController = TextEditingController(text: 'MySecretKey32CharsLongForAES256!');
  String _aesCipher = '';
  String _decrypted = '';
  String _sha256Hash = '';

  void _processCrypto() {
    final text = _inputController.text;
    if (text.isEmpty) return;

    // 1. SHA-256 Hash
    final bytes = utf8.encode(text);
    final digest = crypto.sha256.convert(bytes);

    // 2. AES-256 Symmetric Encryption
    String cipherText = '';
    String clearResult = '';
    try {
      final key = enc.Key.fromUtf8(_keyController.text.padRight(32, '0').substring(0, 32));
      final iv = enc.IV.fromLength(16);
      final encrypter = enc.Encrypter(enc.AES(key));

      final encrypted = encrypter.encrypt(text, iv: iv);
      cipherText = encrypted.base64;
      clearResult = encrypter.decrypt64(cipherText, iv: iv);
    } catch (e) {
      cipherText = 'Key error: Requires 32-byte string';
    }

    setState(() {
      _sha256Hash = digest.toString();
      _aesCipher = cipherText;
      _decrypted = clearResult;
    });
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: SingleChildScrollView(
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            const Text('Encryption & Integrity Utility', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
            const SizedBox(height: 12),
            TextField(
              controller: _inputController,
              decoration: const InputDecoration(labelText: 'Plaintext input', border: OutlineInputBorder()),
            ),
            const SizedBox(height: 10),
            TextField(
              controller: _keyController,
              decoration: const InputDecoration(labelText: '32-Character AES Key', border: OutlineInputBorder()),
            ),
            const SizedBox(height: 10),
            ElevatedButton(onPressed: _processCrypto, child: const Text('Compute Transformations')),
            const SizedBox(height: 16),
            Card(
              child: Padding(
                padding: const EdgeInsets.all(12.0),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    const Text('SHA-256 Digest:', style: TextStyle(fontWeight: FontWeight.bold, color: Color(0xFF00FF66))),
                    SelectableText(_sha256Hash.isEmpty ? '-' : _sha256Hash),
                    const Divider(),
                    const Text('AES-256 Ciphertext (Base64):', style: TextStyle(fontWeight: FontWeight.bold, color: Color(0xFF58A6FF))),
                    SelectableText(_aesCipher.isEmpty ? '-' : _aesCipher),
                    const Divider(),
                    const Text('Decrypted Verification:', style: TextStyle(fontWeight: FontWeight.bold)),
                    SelectableText(_decrypted.isEmpty ? '-' : _decrypted),
                  ],
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

// 5. LOCAL SIEM & EVENT CORRELATOR
// ==========================================
class SiemLogScreen extends StatefulWidget {
  const SiemLogScreen({super.key});

  @override
  State<SiemLogScreen> createState() => _SiemLogScreenState();
}

class _SiemLogScreenState extends State<SiemLogScreen> {
  final List<Map<String, dynamic>> _logs = [];
  final List<String> _alerts = [];

  void _ingestEvent(String type, String ip, String message) {
    final entry = {
      'timestamp': DateTime.now(),
      'type': type,
      'ip': ip,
      'message': message,
    };

    setState(() {
      _logs.insert(0, entry);
      _evaluateSiemRules();
    });
  }

  void _evaluateSiemRules() {
    _alerts.clear();
    final now = DateTime.now();
    final Map<String, int> failedLoginsByIp = {};

    for (var log in _logs) {
      final logTime = log['timestamp'] as DateTime;
      // Sliding correlation window: last 60 seconds
      if (now.difference(logTime).inSeconds <= 60) {
        if (log['type'] == 'AUTH_FAIL') {
          failedLoginsByIp[log['ip']] = (failedLoginsByIp[log['ip']] ?? 0) + 1;
        }
      }
    }

    // SIEM Detection Rule: > 3 failed auth attempts within 1 min
    failedLoginsByIp.forEach((ip, count) {
      if (count >= 3) {
        _alerts.add('SIEM ALERT: Potential Brute-Force attack detected from $ip ($count events/min)');
      }
    });
  }

  @override
  Widget build(BuildContext context) {
    return Padding(
      padding: const EdgeInsets.all(16.0),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          const Text('SIEM Event Monitor & Rule Engine', style: TextStyle(fontSize: 20, fontWeight: FontWeight.bold)),
          const SizedBox(height: 10),
          Wrap(
            spacing: 8,
            children: [
              ActionChip(
                label: const Text('+ Auth Fail (192.168.1.50)'),
                onPressed: () => _ingestEvent('AUTH_FAIL', '192.168.1.50', 'Invalid password supplied'),
              ),
              ActionChip(
                label: const Text('+ Auth Success'),
                onPressed: () => _ingestEvent('AUTH_OK', '192.168.1.50', 'User logged in via 2FA'),
              ),
              ActionChip(
                label: const Text('+ Firewall Block'),
                onPressed: () => _ingestEvent('FW_DROP', '10.0.0.12', 'Port 445 packet dropped'),
              ),
            ],
          ),
          const SizedBox(height: 10),
          if (_alerts.isNotEmpty)
            Container(
              padding: const EdgeInsets.all(8),
              decoration: BoxDecoration(
                color: Colors.red.withOpacity(0.2),
                border: Border.all(color: Colors.redAccent),
                borderRadius: BorderRadius.circular(8),
              ),
              child: Column(
                children: _alerts
                    .map((a) => Row(children: [
                          const Icon(Icons.crisis_alert, color: Colors.redAccent),
                          const SizedBox(width: 8),
                          Expanded(child: Text(a, style: const TextStyle(color: Colors.redAccent))),
                        ]))
                    .toList(),
              ),
            ),
          const SizedBox(height: 10),
          Expanded(
            child: ListView.builder(
              itemCount: _logs.length,
              itemBuilder: (context, idx) {
                final log = _logs[idx];
                final time = DateFormat('HH:mm:ss').format(log['timestamp']);
                return Card(
                  child: ListTile(
                    dense: true,
                    leading: Text(time, style: const TextStyle(color: Colors.grey)),
                    title: Text('${log['type']} [${log['ip']}]'),
                    subtitle: Text(log['message']),
                  ),
                );
              },
            ),
          )
        ],
      ),
    );
  }
}
